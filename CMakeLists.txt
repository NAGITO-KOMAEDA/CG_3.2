cmake_minimum_required(VERSION 3.20)
project(SponzaCSM LANGUAGES CXX)

add_executable(SponzaCSM WIN32 src/main.cpp)
target_compile_features(SponzaCSM PRIVATE cxx_std_17)
target_compile_definitions(SponzaCSM PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(SponzaCSM PRIVATE d3d11 dxgi d3dcompiler)
if(MSVC)
    set_property(TARGET SponzaCSM PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(SponzaCSM PRIVATE /W4 /EHsc /utf-8)
endif()
