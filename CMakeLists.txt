cmake_minimum_required(VERSION 3.20)
project(CG_Homework_4 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(CG_Homework_4 WIN32 src/main.cpp)
target_include_directories(CG_Homework_4 PRIVATE src)
target_compile_definitions(CG_Homework_4 PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(CG_Homework_4 PRIVATE d3d11 dxgi d3dcompiler)
if(MSVC)
  target_compile_options(CG_Homework_4 PRIVATE /W4 /EHsc /utf-8)
endif()
set_target_properties(CG_Homework_4 PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")

add_executable(culling_tests tests/culling_tests.cpp)
target_include_directories(culling_tests PRIVATE src)
if(MSVC)
  target_compile_options(culling_tests PRIVATE /W4 /EHsc /utf-8)
endif()
enable_testing()
add_test(NAME culling_tests COMMAND culling_tests)
